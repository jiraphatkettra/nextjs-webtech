คำศัพท์
ความหมายและตัวอย่างในใบงานนี้
Controlled Input
อินพุตที่ค่าถูกกำหนดจาก State ผ่าน value และแจ้งการเปลี่ยนแปลงกลับผ่าน onChange ในใบงานนี้ใช้กับทุกช่องกรอกของ CourseForm
Computed Property Name
การใช้ชื่อคีย์ที่คำนวณขณะทำงานภายในวงเล็บเหลี่ยม เช่น [name]: value ทำให้ใช้ฟังก์ชันเดียวรองรับทุกฟิลด์ได้
FormEvent
ชนิดของเหตุการณ์การส่งฟอร์มใน TypeScript ใช้กับ onSubmit ของ <form>
preventDefault
เมธอดที่ยับยั้งพฤติกรรมเริ่มต้นของเบราว์เซอร์ ในที่นี้คือการโหลดหน้าใหม่เมื่อส่งฟอร์ม
Validation
การตรวจสอบความถูกต้องของข้อมูลก่อนนำไปบันทึก ในใบงานนี้ตรวจรหัสวิชา ชื่อวิชา และหน่วยกิต
Error State
State ที่เก็บข้อความแจ้งเตือนของแต่ละฟิลด์ ใช้ตัดสินใจว่าจะแสดงข้อความใดใต้อินพุต
Immutable Update
การสร้างค่าใหม่แทนการแก้ไขค่าเดิม ในใบงานนี้ใช้ Spread Operator กับ filter และ map
crypto.randomUUID
เมธอดมาตรฐานของเบราว์เซอร์ที่สร้างรหัสสุ่มไม่ซ้ำกันในรูปแบบ UUID ใช้กำหนดรหัสให้รายวิชาที่เพิ่มใหม่
Callback Props
ฟังก์ชันที่ Parent Component ส่งลงไปให้ Child Component เรียกกลับเมื่อเกิดเหตุการณ์ เช่น onSave onDelete และ onEdit
Dynamic Route
เส้นทางที่มีบางส่วนของ URL เปลี่ยนแปลงได้ สร้างด้วยโฟลเดอร์ชื่อ [id] รองรับ /courses/CS101 และ /courses/CS102 ด้วยไฟล์เดียว
params
อ็อบเจกต์ที่บรรจุค่าของ Dynamic Segment ตั้งแต่ Next.js เวอร์ชัน 15 เป็นต้นไป เป็น Promise จึงต้องใช้ await ก่อนอ่านค่า
notFound
ฟังก์ชันจาก next/navigation ที่สั่งให้ Next.js แสดงหน้า 404 ใช้เมื่อไม่พบข้อมูลตามพารามิเตอร์
generateMetadata
ฟังก์ชันที่ Next.js เรียกเพื่อสร้าง Metadata ของหน้าจากข้อมูลจริง ใช้แทน metadata เมื่อชื่อหน้าขึ้นกับพารามิเตอร์
Link
Component จาก next/link ที่เปลี่ยนหน้าโดยไม่โหลดเอกสารใหม่ทั้งหน้า
แผนภาพการทำงานของหน้า /courses และหน้ารายละเอียด

// src/reducers/coursesReducer.ts
import type { Course } from "@/types/course";
export type CoursesAction =
  | { type: "created"; course: Course }
  | { type: "deleted"; id: string }
  | { type: "updated"; id: string; course: Course };
export function coursesReducer(
  state: Course[],
  action: CoursesAction
): Course[] {
  switch (action.type) {
    case "created":
      return [...state, action.course];
    case "deleted":
      return state.filter((course) => course.id !== action.id);
    case "updated":
      return state.map((course) =>
        course.id === action.id ? action.course : course
      );
    default:
      return state;
  }
}
// src/components/CourseExplorer.tsx
"use client";
import { useReducer, useState, type ChangeEvent } from "react";
import type { Course } from "@/types/course";
import CourseCard from "@/components/CourseCard";
import CourseForm, { type CourseDraft } from "@/components/CourseForm";
import { coursesReducer } from "@/reducers/coursesReducer";
type CourseExplorerProps = {
  initialCourses: Course[];
};
export default function CourseExplorer(
  { initialCourses }: CourseExplorerProps
) {
  const [courses, dispatch] = useReducer(coursesReducer, initialCourses);
  const [keyword, setKeyword] = useState("");
  const [editingId, setEditingId] = useState<string | null>(null);
  function toCourse(id: string, draft: CourseDraft): Course {
    return {
      id,
      code: draft.code.trim(),
      name: draft.name.trim(),
      credit: Number(draft.credit),
      instructor: draft.instructor.trim(),
    };
  }
  function handleSave(draft: CourseDraft) {
    if (editingId === null) {
      dispatch({ type: "created", course: toCourse(crypto.randomUUID(), draft) });
      return;
    }
    dispatch({ type: "updated", id: editingId, course: toCourse(editingId, draft) });
    setEditingId(null);
  }
  const editingCourse = courses.find((course) => course.id === editingId);
  const searchText = keyword.trim().toLowerCase();
  const visibleCourses = courses.filter((course) =>
    course.name.toLowerCase().includes(searchText) ||
    course.code.toLowerCase().includes(searchText)
  );
  function handleKeywordChange(event: ChangeEvent<HTMLInputElement>) {
    setKeyword(event.target.value);
  }
  return (
    <div>
      <input id="keyword" placeholder="ค้นหารายวิชา" value={keyword}
             onChange={handleKeywordChange} />
      <CourseForm
        key={editingId ?? "new"}
        initialCourse={editingCourse}
        onSave={handleSave}
        onCancel={() => setEditingId(null)}
      />
      {visibleCourses.length === 0 ? (
        <p>ไม่พบรายวิชาที่ตรงกับคำค้น</p>
      ) : (
        visibleCourses.map((course) => (
          <CourseCard
            key={course.id}
            course={course}
            onEdit={() => setEditingId(course.id)}
            onDelete={() => dispatch({ type: "deleted", id: course.id })}
          />
        ))
      )}
    </div>
  );
}

// src/lib/coursesSlice.ts
import { createSlice, type PayloadAction } from "@reduxjs/toolkit";
import type { Course } from "@/types/course";
const coursesSlice = createSlice({
  name: "courses",
  initialState: [] as Course[],
  reducers: {
    initialized(state, action: PayloadAction<Course[]>) {
      return action.payload;
    },
    created(state, action: PayloadAction<Course>) {
      state.push(action.payload);
    },
    deleted(state, action: PayloadAction<string>) {
      return state.filter((course) => course.id !== action.payload);
    },
    updated(state, action: PayloadAction<Course>) {
      const index = state.findIndex((course) => course.id === action.payload.id);
      if (index !== -1) {
        state[index] = action.payload;
      }
    },
  },
});
export const { initialized, created, deleted, updated } = coursesSlice.actions;
export default coursesSlice.reducer;

"use client";

import type { Course } from "@/types/course";
import CourseCard from "@/components/CourseCard";
import CourseForm, { type CourseDraft } from "@/components/CourseForm";
import { useAppDispatch, useAppSelector, useAppStore } from "@/lib/hooks";
import { initialized, created, deleted, updated } from "@/lib/coursesSlice";
type CourseExplorerProps = {
  initialCourses: Course[];
};
export default function CourseExplorer(
  { initialCourses }: CourseExplorerProps
) {
  const store = useAppStore();
  const isInitialized = useRef(false);
  if (!isInitialized.current) {
    store.dispatch(initialized(initialCourses));
    isInitialized.current = true;
  }
  const courses = useAppSelector((state) => state.courses);
  const dispatch = useAppDispatch();
  const [keyword, setKeyword] = useState("");

  function toCourse(id: string, draft: CourseDraft): Course {

      credit: Number(draft.credit),
      instructor: draft.instructor.trim(),
    };
  }
  function handleSave(draft: CourseDraft) {
    if (editingId === null) {
      dispatch(created(toCourse(crypto.randomUUID(), draft)));
      return;
    }
    dispatch(updated(toCourse(editingId, draft)));
    setEditingId(null);
  }
  const editingCourse = courses.find((course) => course.id === editingId);
  const searchText = keyword.trim().toLowerCase();
  const visibleCourses = courses.filter((course) =>
    course.name.toLowerCase().includes(searchText) ||
    course.code.toLowerCase().includes(searchText)
  );
  function handleKeywordChange(event: ChangeEvent<HTMLInputElement>) {
    setKeyword(event.target.value);
  }
  return (
    <div>
      <input id="keyword" placeholder="ค้นหารายวิชา" value={keyword}
             onChange={handleKeywordChange} />
      <CourseForm
        key={editingId ?? "new"}
        initialCourse={editingCourse}
        onSave={handleSave}
        onCancel={() => setEditingId(null)}
      />
      {visibleCourses.length === 0 ? (

            key={course.id}
            course={course}
            onEdit={() => setEditingId(course.id)}
